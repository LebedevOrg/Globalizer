# requirements.txt для Globalizer

# Только библиотеки, необходимые самому Globalizer во время работы:
#   * ядро Python-части (встроенный Python, PYProblem, globalizer.optimize,
#     globalizer.optuna) - numpy;
#   * встроенная визуализация lib/plotters (графики при IsPlot = true) -

#   requirements_build.txt   - сборка Python-интерфейса (модуль PYDGlobalizer);

#                              (examples/compare_optimizers.py) и проверки совместимости;
#   third_party/Problems/Problems/iOptProblemSimple/requirements.txt
#                            - задача iOptProblemSimple (requests, tqdm).

# ---------- Ядро: numpy ----------
# numpy 2.x требует Python >=3.10; для 3.8/3.9 остаёмся на 1.24.x.
numpy>=1.24,<2.0 ; python_version < "3.10"
numpy>=1.26,<3.0 ; python_version >= "3.10"

# ---------- Визуализация (lib/plotters) ----------
# matplotlib 3.8+ тянет contourpy автоматически (не пиним его!).
matplotlib>=3.6,<4

# plotters.py: MLPRegressor (аппроксимация поверхности по точкам испытаний).
# scikit-learn сам зависит от scipy (scipy.interpolate тоже нужен plotters.py),
# поэтому отдельной строки для scipy здесь нет.
# scikit-learn: 1.4+ требует Python >=3.9; на 3.8 берём 1.3.x
scikit-learn>=1.2,<1.4 ; python_version < "3.9"
scikit-learn>=1.2,<2   ; python_version >= "3.9"
