# ============================================================================
# requirements_build.txt — зависимости для СБОРКИ PYGlobalizer Python module
# ============================================================================
#
# НАЗНАЧЕНИЕ:
# -----------
# Этот файл содержит минимальные зависимости для КОМПИЛЯЦИИ C++ модуля
# PYGlobalizer с помощью CMake + pybind11.
#
# ИСПОЛЬЗОВАНИЕ:
# --------------
# 1. Создайте виртуальное окружение (рекомендуется):
#    python3.11 -m venv venv
#    source venv/bin/activate  # Linux/macOS
#    venv\Scripts\activate     # Windows
#
# 2. Установите зависимости:
#    pip install -r requirements_build.txt
#
# 3. Соберите проект:
#    mkdir -p build && cd build
#    cmake .. -DPYGlobalizer=ON -DCMAKE_BUILD_TYPE=Release
#    cmake --build . --config Release -j$(nproc)
#
# СИСТЕМНЫЕ ТРЕБОВАНИЯ:
# ---------------------
# - Python >= 3.11 (проверено CMakeLists.txt:find_package(Python3 3.11))
# - C++17 compatible compiler (GCC >= 7, Clang >= 5, MSVC >= 2017)
# - CMake >= 3.15
# - OpenMP (опционально, для параллелизма)
#
# ============================================================================

# ----------------------------------------------------------------------------
# 1. PYBIND11 — обязательная зависимость для биндингов Python ↔ C++
# ----------------------------------------------------------------------------
# pybind11 требуется ДО сборки, т.к. CMake ищет его через find_package(pybind11).
# Минимальная версия: 2.11 (поддержка py::gil_scoped_acquire, py::object)
# Рекомендуется: 2.13+ (улучшенная обработка исключений, лучший typing)

pybind11>=2.13.0,<3.0

# АЛЬТЕРНАТИВА (если pybind11 уже установлен глобально или через submodule):
# Если pybind11 подключен как git submodule в third_party/pybind11,
# эту строку можно закомментировать, CMake найдет его автоматически.

# ----------------------------------------------------------------------------
# 2. NUMPY — требуется для работы с массивами в Python
# ----------------------------------------------------------------------------
# Нужен на этапе сборки для:
# - Заголовков numpy/arrayobject.h (если используются np.ndarray в C++)
# - Runtime dependency для PYProblem.py (type hints с numpy.typing)
#
# ВАЖНО: numpy 2.x требует Python >= 3.10
# Для Python 3.11+ берём numpy >= 2.0 (новый ABI, быстрее)

numpy>=2.0.0,<3.0 ; python_version >= "3.11"

# Для более старых Python (если нужно):
# numpy>=1.24,<2.0 ; python_version < "3.10"
# numpy>=1.26,<3.0 ; python_version >= "3.10" and python_version < "3.11"

# ----------------------------------------------------------------------------
# 3. SETUPTOOLS и WHEEL — для сборки Python-пакетов (pip install -e .)
# ----------------------------------------------------------------------------
# Если планируете делать pip install -e . (editable install), нужны:

setuptools>=65.0
wheel>=0.40.0

# ----------------------------------------------------------------------------
# 4. CMAKE (опционально через pip, если не установлен системно)
# ----------------------------------------------------------------------------
# Если в системе нет CMake >= 3.15, можно установить через pip:
# cmake>=3.15

# НО ЛУЧШЕ установить системный CMake:
# - Ubuntu/Debian: sudo apt install cmake
# - macOS: brew install cmake
# - Windows: scoop install cmake или скачать с cmake.org

# ----------------------------------------------------------------------------
# 5. NINJA (опционально, для ускорения сборки)
# ----------------------------------------------------------------------------
# Ninja — быстрая альтернатива make. Использование:
# cmake .. -GNinja -DPYGlobalizer=ON
# ninja

# ninja>=1.11

# Для загрузки тестовых задач из iOptProblemSimple:
requests>=2.31,<3

# ============================================================================
# ПРОВЕРКА УСТАНОВКИ
# ============================================================================
# После установки проверьте версии:
#
# python3 --version           # Должна быть >= 3.11
# python3 -c "import pybind11; print(pybind11.__version__)"  # >= 2.13.0
# python3 -c "import numpy; print(numpy.__version__)"        # >= 2.0.0
# cmake --version             # >= 3.15
#
# Если всё OK, переходите к сборке (см. выше, раздел ИСПОЛЬЗОВАНИЕ)
#
# ============================================================================
# TROUBLESHOOTING
# ============================================================================
#
# ОШИБКА: "Could NOT find pybind11"
# РЕШЕНИЕ: pip install pybind11>=2.13
#          или добавьте -Dpybind11_DIR=/path/to/pybind11/share/cmake/pybind11
#
# ОШИБКА: "Python.h: No such file or directory"
# РЕШЕНИЕ: Установите python3-dev:
#          Ubuntu/Debian: sudo apt install python3.11-dev
#          Fedora: sudo dnf install python3-devel
#          macOS: brew install python@3.11
#
# ОШИБКА: "undefined reference to `omp_get_num_threads'"
# РЕШЕНИЕ: Установите OpenMP:
#          Ubuntu/Debian: sudo apt install libomp-dev
#          macOS: brew install libomp
#          Windows: уже включен в MSVC
#
# ОШИБКА при импорте: "ImportError: cannot import name 'PYGlobalizer'"
# РЕШЕНИЕ: Убедитесь, что _bin/PYGlobalizer.so (Linux/macOS) или
#          _bin/PYGlobalizer.pyd (Windows) скопирован в PYTHONPATH, либо:
#          export PYTHONPATH=$PYTHONPATH:$(pwd)/_bin
#
